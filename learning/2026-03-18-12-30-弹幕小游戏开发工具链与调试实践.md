    private frameTimes: number[] = [];
    
    update(dt: number) {
        const fps = 1 / dt;
        this.frameTimes.push(fps);

### 2.2 内存调试

**常见内存泄漏场景：**

| 场景 | 问题 | 解决方案 |
|------|------|----------|
| 回调未移除 | 事件监听未清理 | onDestroy中取消订阅 |
| 闭包引用 | 长期持有对象 | 弱引用/及时释放 |
| 数组累积 | 数据不断添加 | 限制长度/分页 |
| 纹理未释放 | 资源未手动释放 | destroy纹理对象 |

### 3.1 分级日志

```typescript
enum LogLevel {
    DEBUG = 0,
    INFO = 1,
    WARN = 2,
    ERROR = 3,
}

class Logger {
    private static level: LogLevel = LogLevel.DEBUG;
    
    static debug(...args: any[]): void {
        if (this.level <= LogLevel.DEBUG) {
            console.debug('[DEBUG]', ...args);
        }
    }
    
    static info(...args: any[]): void {
        if (this.level <= LogLevel.INFO) {
            console.info('[INFO]', ...args);
        }
    }
    
    static warn(...args: any[]): void {
        if (this.level <= LogLevel.WARN) {
            console.warn('[WARN]', ...args);
        }
    }
    
    static error(...args: any[]): void {
        if (this.level <= LogLevel.ERROR) {
            console.error('[ERROR]', ...args);
        }
    }
}
```

### 4.2 性能监控

```typescript
// 关键指标监控
class PerformanceMonitor {
    private metrics: Map<string, number[]> = new Map();
    
    // 记录耗时
    recordTime(label: string, duration: number): void {
        if (!this.metrics.has(label)) {
            this.metrics.set(label, []);
        }
        this.metrics.get(label)!.push(duration);
        
        // 上报
        if (this.metrics.get(label)!.length >= 10) {
            this.reportMetric(label);
        }
    }
    
    // 上报指标
    private reportMetric(label: string): void {
        const times = this.metrics.get(label)!;
        const avg = times.reduce((a, b) => a + b) / times.length;
        
        fetch('/api/metrics', {
            method: 'POST',
            body: JSON.stringify({ label, avg, count: times.length })
        });
        
        this.metrics.set(label, []);
    }
}
```
