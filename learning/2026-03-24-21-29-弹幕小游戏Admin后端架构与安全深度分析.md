### 4.4 ⚠️ 安全风险：字段类型全部为string

**问题：**
- Count/Total/GiftCount 是数字但存储为string
- 需要`strconv.Atoi()`转换，可能panic
- 无范围校验（负数、超大数）

**防御建议：**
```go
// 当前代码（有风险）
count, _ := strconv.Atoi(msg.Count)

// 防御性编程
func safeAtoi(s string, defaultVal int) int {
    if s == "" {
        return defaultVal
    }
    i, err := strconv.Atoi(s)
    if err != nil || i < 0 {
        return defaultVal
    }
    return i
}
```

## 五、Admin后端安全加固方案（本轮重要产出）

### 5.1 紧急需要添加的中间件

**1. Rate Limiting中间件（rate_limit.go）：**

```go
package middleware

import (
    "net/http"
    "sync"
    "time"
    "golang.org/x/time/rate"
)

type RateLimiter struct {
    limiters map[string]*rate.Limiter
    mu       sync.RWMutex
    rps      rate.Limit // 请求/秒
    burst    int        // 突发容量
}

func NewRateLimiter(rps float64, burst int) *RateLimiter {
    return &RateLimiter{
        limiters: make(map[string]*rate.Limiter),
        rps:      rate.Limit(rps),
        burst:    burst,
    }
}

func (rl *RateLimiter) getLimiter(key string) *rate.Limiter {
    rl.mu.Lock()
    defer rl.mu.Unlock()
    
    if _, ok := rl.limiters[key]; !ok {
        rl.limiters[key] = rate.NewLimiter(rl.rps, rl.burst)
    }
    return rl.limiters[key]
}

// 登录接口：5次/分钟
// 普通接口：100次/秒
func LoginRateLimit() gin.HandlerFunc {
    limiter := NewRateLimiter(5.0/60.0, 5) // 5次/分钟
    return func(c *gin.Context) {
        ip := c.ClientIP()
        if !limiter.getLimiter(ip).Allow() {
            c.JSON(http.StatusTooManyRequests, gin.H{
                "code": 429,
                "msg":  "请求过于频繁，请稍后再试",
            })
            c.Abort()
            return
        }
        c.Next()
    }
}
```

**2. CORS中间件（cors.go）：**

```go
func CORS() gin.HandlerFunc {
    return func(c *gin.Context) {
        origin := c.Request.Header.Get("Origin")
        allowedOrigins := map[string]bool{
            "https://admin.example.com": true,
        }
        
        if allowedOrigins[origin] {
            c.Writer.Header().Set("Access-Control-Allow-Origin", origin)
            c.Writer.Header().Set("Access-Control-Allow-Credentials", "true")
            c.Writer.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
            c.Writer.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
        }
        
        if c.Request.Method == "OPTIONS" {
            c.AbortWithStatus(http.StatusNoContent)
            return
        }
        c.Next()
    }
}
```

**3. RequestID中间件：**

```go
import "github.com/google/uuid"

func RequestID() gin.HandlerFunc {
    return func(c *gin.Context) {
        requestID := c.GetHeader("X-Request-ID")
        if requestID == "" {
            requestID = uuid.New().String()
        }
        c.Set("request_id", requestID)
        c.Writer.Header().Set("X-Request-ID", requestID)
        c.Next()
    }
}
```

**4. Recovery中间件（增强版）：**

```go
func Recovery() gin.HandlerFunc {
    return func(c *gin.Context) {
        defer func() {
            if err := recover(); err != nil {
                requestID, _ := c.Get("request_id")
                zap.L().Error("panic recovered",
                    zap.Any("error", err),
                    zap.String("request_id", requestID.(string)),
                    zap.String("path", c.Request.URL.Path),
                    zap.String("method", c.Request.Method),
                )
                c.JSON(http.StatusInternalServerError, gin.H{
                    "code": 500,
                    "msg":  "服务器内部错误",
                    "request_id": requestID,
                })
                c.Abort()
            }
        }()
        c.Next()
    }
}
```

### 5.2 Gin中间件注册顺序

```go
func SetupRouter() *gin.Engine {
    r := gin.New()
    
    // 1. Recovery（最先，防止崩溃）
    r.Use(middleware.Recovery())
    
    // 2. RequestID（请求追踪）
    r.Use(middleware.RequestID())
    
    // 3. Logger（日志）
    r.Use(zapLogger())
    
    // 4. CORS（跨域）
    r.Use(middleware.CORS())
    
    // 5. Rate Limiting（限流，按需）
    // 登录接口单独限流
    loginGroup := r.Group("/api/login")
    loginGroup.Use(middleware.LoginRateLimit())
    
    // 6. JWT认证（需要认证的接口）
    apiGroup := r.Group("/api")
    apiGroup.Use(middleware.JWTAuth())
    
    return r
}
```
