## 二、高并发弹幕服务器设计（Go + Redis）

### 2.1 为什么选择Go

弹幕服务器的核心挑战是**高并发+高吞吐+低延迟**：
- 单个热门直播间弹幕可达 **500-1000条/秒**
- 抖音API推送延迟要求 < **50ms**
- 需要支撑 **10000+** 并发连接

Go的优势：
- **goroutine**：轻量级并发，10万连接仅需少量内存
- **channel**：天然适合消息队列
- **GC优化**：Go 1.21延迟降低50%
- **Redis客户端**：成熟生态（go-redis/redis.v9）

### 2.2 连接管理器设计

```go
// servers/logic/control/connection_manager.go

package control

import (
    "sync"
    "sync/atomic"
    "time"
)

// ClientConnection 单个客户端连接
type ClientConnection struct {
    UserID     string
    RoomID     string
    WSConn     interface{}  // WebSocket连接接口
    JoinTime   time.Time
    LastActive time.Time
    IsAlive    atomic.Bool
    SendChan   chan []byte  // 发送队列
}

// ConnectionManager 连接管理器
type ConnectionManager struct {
    mu           sync.RWMutex
    clients      map[string]*ClientConnection  // userID -> connection
    roomClients  map[string]map[string]*ClientConnection  // roomID -> (userID -> connection)
    totalConn    atomic.Int64
    maxConn      int64
}

func NewConnectionManager(maxConn int64) *ConnectionManager {
    return &ConnectionManager{
        clients:     make(map[string]*ClientConnection),
        roomClients: make(map[string]map[string]*ClientConnection),
        maxConn:     maxConn,
    }
}

// Add 添加连接
func (cm *ConnectionManager) Add(client *ClientConnection) error {
    if cm.totalConn.Load() >= cm.maxConn {
        return ErrMaxConnectionsReached
    }
    
    cm.mu.Lock()
    defer cm.mu.Unlock()
    
    // 维护全局映射
    cm.clients[client.UserID] = client
    
    // 维护房间映射
    if _, ok := cm.roomClients[client.RoomID]; !ok {
        cm.roomClients[client.RoomID] = make(map[string]*ClientConnection)
    }
    cm.roomClients[client.RoomID][client.UserID] = client
    
    cm.totalConn.Add(1)
    return nil
}

// Remove 移除连接
func (cm *ConnectionManager) Remove(userID string) {
    cm.mu.Lock()
    defer cm.mu.Unlock()
    
    client, ok := cm.clients[userID]
    if !ok {
        return
    }
    
    // 从全局映射删除
    delete(cm.clients, userID)
    
    // 从房间映射删除
    if roomClients, ok := cm.roomClients[client.RoomID]; ok {
        delete(roomClients, userID)
        if len(roomClients) == 0 {
            delete(cm.roomClients, client.RoomID)
        }
    }
    
    cm.totalConn.Add(-1)
}

// BroadcastToRoom 广播消息到房间
func (cm *ConnectionManager) BroadcastToRoom(roomID string, msg []byte) {
    cm.mu.RLock()
    roomClients, ok := cm.roomClients[roomID]
    cm.mu.RUnlock()
    
    if !ok {
        return
    }
    
    // 异步广播，避免阻塞
    for _, client := range roomClients {
        select {
        case client.SendChan <- msg:
        default:
            // 通道满，跳过（防止慢客户端阻塞）
        }
    }
}

// GetRoomClientCount 获取房间在线人数
func (cm *ConnectionManager) GetRoomClientCount(roomID string) int {
    cm.mu.RLock()
    defer cm.mu.RUnlock()
    
    if roomClients, ok := cm.roomClients[roomID]; ok {
        return len(roomClients)
    }
    return 0
}
```

### 2.3 Redis实时排行榜设计

弹幕游戏排行榜需求：
- **弹幕积分榜**：按弹幕发送量排名（反映互动活跃度）
- **礼物价值榜**：按累计礼物金额排名（核心变现指标）
- **今日/本周/本月**：多时间维度

```go
// servers/logic/model/model_rank_redis.go（扩展版）

package model

import (
    "context"
    "fmt"
    "time"
    
    "github.com/redis/go-redis/v9"
)

const (
    RankKeyDanmaku     = "rank:danmaku:%s:%s"  // rank:danmaku:roomID:day
    RankKeyGift        = "rank:gift:%s:%s"     // rank:gift:roomID:day
    RankKeyGiftAmount  = "rank:gift:amount:%s:%s" // rank:gift:amount:roomID:day
    RankExpireDays    = 7                       // 排行榜保留7天
)

// RankType 排行榜类型
type RankType int

const (
    RankTypeDanmaku RankType = iota
    RankTypeGiftCount
    RankTypeGiftAmount
)

// RankEntry 排行榜条目
type RankEntry struct {
    UserID     string  `json:"user_id"`
    Nickname   string  `json:"nickname"`
    Score      int64   `json:"score"`      // 弹幕数 or 礼物数
    Amount     int64   `json:"amount"`     // 礼物金额（仅礼物榜）
    Rank       int     `json:"rank"`
}

// RankRedis 排行榜Redis操作
type RankRedis struct {
    rdb *redis.Client
    ctx context.Context
}

// NewRankRedis 创建排行榜Redis客户端
func NewRankRedis(addr, password string, db int) *RankRedis {
    rdb := redis.NewClient(&redis.Options{
        Addr:     addr,
        Password: password,
        DB:       db,
        PoolSize: 100,  // 连接池大小
    })
    
    ctx := context.Background()
    return &RankRedis{rdb: rdb, ctx: ctx}
}

// IncrDanmakuCount 增加弹幕计数
func (r *RankRedis) IncrDanmakuCount(roomID, userID string, count int64) error {
    key := r.getKey(RankKeyDanmaku, roomID, time.Now().Format("2006-01-02"))
    
    pipe := r.rdb.Pipeline()
    pipe.ZIncrBy(r.ctx, key, float64(count), userID)
    pipe.Expire(r.ctx, key, RankExpireDays*24*time.Hour)
    
    _, err := pipe.Exec(r.ctx)
    return err
}

// IncrGiftStat 增加礼物统计
func (r *RankRedis) IncrGiftStat(roomID, userID string, giftCount int, giftAmount int64) error {
    day := time.Now().Format("2006-01-02")
    
    pipe := r.rdb.Pipeline()
    
    // 礼物次数排行
    countKey := r.getKey(RankKeyGift, roomID, day)
    pipe.ZIncrBy(r.ctx, countKey, float64(giftCount), userID)
    pipe.Expire(r.ctx, countKey, RankExpireDays*24*time.Hour)
    
    // 礼物金额排行
    amountKey := r.getKey(RankKeyGiftAmount, roomID, day)
    pipe.ZIncrBy(r.ctx, amountKey, float64(giftAmount), userID)
    pipe.Expire(r.ctx, amountKey, RankExpireDays*24*time.Hour)
    
    _, err := pipe.Exec(r.ctx)
    return err
}

// GetTopN 获取TopN排行榜
func (r *RankRedis) GetTopN(roomID string, rankType RankType, topN int) ([]RankEntry, error) {
    day := time.Now().Format("2006-01-02")
    
    var key string
    switch rankType {
    case RankTypeDanmaku:
        key = r.getKey(RankKeyDanmaku, roomID, day)
    case RankTypeGiftCount:
        key = r.getKey(RankKeyGift, roomID, day)
    case RankTypeGiftAmount:
        key = r.getKey(RankKeyGiftAmount, roomID, day)
    default:
        return nil, fmt.Errorf("unknown rank type")
    }
    
    // ZREVRANGE with scores，返回排名前N的用户
    results, err := r.rdb.ZRevRangeWithScores(r.ctx, key, 0, int64(topN-1)).Result()
    if err != nil {
        return nil, err
    }
    
    entries := make([]RankEntry, 0, len(results))
    for i, z := range results {
        entries = append(entries, RankEntry{
            UserID: z.Member.(string),
            Score:  int64(z.Score),
            Rank:   i + 1,
        })
    }
    
    return entries, nil
}

// GetUserRank 获取用户排名
func (r *RankRedis) GetUserRank(roomID, userID string, rankType RankType) (int64, error) {
    day := time.Now().Format("2006-01-02")
    
    var key string
    switch rankType {
    case RankTypeDanmaku:
        key = r.getKey(RankKeyDanmaku, roomID, day)
    case RankTypeGiftCount:
        key = r.getKey(RankKeyGift, roomID, day)
    case RankTypeGiftAmount:
        key = r.getKey(RankKeyGiftAmount, roomID, day)
    default:
        return 0, fmt.Errorf("unknown rank type")
    }
    
    // ZREVRANK返回0-based排名
    rank, err := r.rdb.ZRevRank(r.ctx, key, userID).Result()
    if err != nil {
        if err == redis.Nil {
            return -1, nil  // 用户不在榜上
        }
        return 0, err
    }
    
    return rank + 1, nil  // 转换为1-based
}

func (r *RankRedis) getKey(template, roomID, day string) string {
    return fmt.Sprintf(template, roomID, day)
}
```

```go
// CooldownManager 冷却管理器
type CooldownManager struct {
    danmakuCooldown map[string]time.Time  // userID -> 上次弹幕时间
    giftCooldown    map[string]time.Time  // userID -> 上次礼物时间
    likeCount       map[string][]time.Time // userID -> 最近点赞时间列表
    lock            sync.RWMutex
}

// NewCooldownManager 创建冷却管理器
func NewCooldownManager() *CooldownManager {
    return &CooldownManager{
        danmakuCooldown: make(map[string]time.Time),
        giftCooldown:    make(map[string]time.Time),
        likeCount:       make(map[string][]time.Time),
    }
}

// CheckAndSetDanmakuCooldown 检查并设置弹幕冷却
func (c *CooldownManager) CheckAndSetDanmakuCooldown(userID string) bool {
    c.lock.Lock()
    defer c.lock.Unlock()
    
    lastTime, ok := c.danmakuCooldown[userID]
    if ok && time.Since(lastTime) < time.Second*DanmakuCooldown {
        return false
    }
    
    c.danmakuCooldown[userID] = time.Now()
    return true
}

// CheckAndSetGiftCooldown 检查并设置礼物冷却
func (c *CooldownManager) CheckAndSetGiftCooldown(userID string) bool {
    c.lock.Lock()
    defer c.lock.Unlock()
    
    lastTime, ok := c.giftCooldown[userID]
    if ok && time.Since(lastTime) < time.Second*1 {
        return false
    }
    
    c.giftCooldown[userID] = time.Now()
    return true
}

// AddLikeCount 增加点赞计数
func (c *CooldownManager) AddLikeCount(userID string, count int) int {
    c.lock.Lock()
    defer c.lock.Unlock()
    
    now := time.Now()
    windowStart := now.Add(-time.Second * LikeBurstWindow)
    
    // 清理过期记录
    times := c.likeCount[userID]
    validTimes := make([]time.Time, 0)
    for _, t := range times {
        if t.After(windowStart) {
            validTimes = append(validTimes, t)
        }
    }
    
    // 添加新计数
    for i := 0; i < count; i++ {
        validTimes = append(validTimes, now)
    }
    c.likeCount[userID] = validTimes
    
    return len(validTimes)
}

// ProcessChatMessage 处理弹幕消息
func (c *Controller) ProcessChatMessage(ctx context.Context, dyMsg *DYLiveMsg) error {
    // 1. 消息去重（基于MsgID）
    if c.isMessageProcessed(dyMsg.MsgID) {
        return nil
    }
    
    // 2. 敏感词过滤
    if untils.SensitiveCheck(dyMsg.Content) {
        return nil
    }
    
    // 3. 冷却检查
    if !c.cooldown.CheckAndSetDanmakuCooldown(dyMsg.UserID) {
        return nil
    }
    
    // 4. 动作匹配
    action, ok := c.actionMap.Match(dyMsg.Content)
    if !ok {
        // 无匹配动作，仅记录积分
        c.rankRedis.IncrDanmakuCount(dyMsg.RoomID, dyMsg.UserID, 1)
        return nil
    }
    
    // 5. 获取玩家权限等级
    player := c.GetPlayer(dyMsg.UserID)
    if player == nil {
        return nil
    }
    
    // 6. 权限检查
    if player.Level < action.MinLevel {
        return nil
    }
    
    // 7. 构造游戏帧消息
    frameMsg := &pb.GameFrame{
        Type:      pb.FrameType_ACTION,
        PlayerID:  dyMsg.UserID,
        Action:    action.ActionType,
        Direction: action.Direction,
        SkillID:   int32(action.SkillID),
        Timestamp: dyMsg.Timestamp,
    }
    
    // 8. 发送到UE4
    if err := c.wsPusher.PushToRoom(dyMsg.RoomID, frameMsg); err != nil {
        return fmt.Errorf("push to UE4 failed: %w", err)
    }
    
    // 9. 更新排行榜
    c.rankRedis.IncrDanmakuCount(dyMsg.RoomID, dyMsg.UserID, 1)
    
    // 10. 更新玩家积分
    c.addPlayerScore(dyMsg.UserID, int64(action.Priority*10))
    
    return nil
}

// ProcessGiftMessage 处理礼物消息
func (c *Controller) ProcessGiftMessage(ctx context.Context, giftMsg *DYGiftMsg) error {
    // 1. 消息去重
    if c.isMessageProcessed(giftMsg.Timestamp) {
        return nil
    }
    
    // 2. 冷却检查
    if !c.cooldown.CheckAndSetGiftCooldown(giftMsg.UserID) {
        return nil
    }
    
    // 3. 礼物配置映射
    unitConfig := c.giftMap.Get(giftMsg.GiftID)
    totalCount := unitConfig.Count * giftMsg.GiftCount
    
    // 4. 构造兵种生成消息
    spawnMsg := &pb.SpawnUnit{
        Type:        unitConfig.UnitType,
        Count:       int32(totalCount),
        TargetTeam:  unitConfig.TargetTeam,
        SpawnerID:   giftMsg.UserID,
        SpawnerName: giftMsg.UserNickname,
        Effect:      unitConfig.Effect,
        Timestamp:   giftMsg.Timestamp,
    }
    
    // 5. 广播到房间
    if err := c.wsPusher.BroadcastToRoom(giftMsg.RoomID, spawnMsg); err != nil {
        return fmt.Errorf("broadcast spawn failed: %w", err)
    }
    
    // 6. 更新礼物排行榜
    c.rankRedis.IncrGiftStat(
        giftMsg.RoomID,
        giftMsg.UserID,
        giftMsg.GiftCount,
        giftMsg.TotalAmount/100, // 转换为元
    )
    
    // 7. 触发特效（如果是大型礼物）
    if unitConfig.UnitType == "boss_dino" || unitConfig.UnitType == "special_dino" {
        c.triggerGiftEffect(giftMsg.RoomID, giftMsg.UserNickname, unitConfig.Effect)
    }
    
    return nil
}

// ProcessLikeMessage 处理点赞消息
func (c *Controller) ProcessLikeMessage(ctx context.Context, likeMsg *DYLikeMsg) error {
    // 1. 点赞计数
    totalLikes := c.cooldown.AddLikeCount(likeMsg.UserID, likeMsg.Count)
    
    // 2. Burst检测
    if totalLikes >= LikeBurstThreshold {
        burstMsg := &pb.LikeBurst{
            UserID:    likeMsg.UserID,
            Count:     int32(totalLikes),
            Timestamp: likeMsg.Timestamp,
        }
        c.wsPusher.BroadcastToRoom(likeMsg.RoomID, burstMsg)
    }
    
    // 3. 普通点赞累积BUFF
    buffMsg := &pb.BuffApply{
        BuffType:  "like_buff",
        Stack:     int32(totalLikes / 5),
        MaxStack:  10,
        Timestamp: likeMsg.Timestamp,
    }
    c.wsPusher.PushToRoom(likeMsg.RoomID, buffMsg)
    
    return nil
}

// ============ 辅助方法 ============

var processedMsgsMu sync.Mutex
var processedMsgs = make(map[string]time.Time)

// isMessageProcessed 检查消息是否已处理（用于去重）
func (c *Controller) isMessageProcessed(msgID interface{}) bool {
    processedMsgsMu.Lock()
    defer processedMsgsMu.Unlock()
    
    key := fmt.Sprintf("%v", msgID)
    if _, ok := processedMsgs[key]; ok {
        return true
    }
    
    processedMsgs[key] = time.Now()
    
    // 清理过期记录（1小时前的）
    for k, t := range processedMsgs {
        if time.Since(t) > time.Hour {
            delete(processedMsgs, k)
        }
    }
    
    return false
}
```
