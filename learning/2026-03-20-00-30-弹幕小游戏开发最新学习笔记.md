
### 3.3 典型实现示例

```javascript
// 弹幕理解 Agent
class DanmakuUnderstandingAgent {
  constructor() {
    this.intentClassifier = new IntentClassifier();
    this.sentimentAnalyzer = new SentimentAnalyzer();
  }

  async process(danmakuText) {
    // 1. 意图分类
    const intent = await this.intentClassifier.predict(danmakuText);
    
    // 2. 情感分析
    const sentiment = await this.sentimentAnalyzer.analyze(danmakuText);
    
    // 3. 参数提取
    const params = this.extractParams(danmakuText, intent);
    
    return {
      intent,      // 'move', 'attack', 'interact', etc.
      sentiment,   // 'positive', 'neutral', 'negative'
      params,      // { direction: 'left', power: 5 }
      confidence: Math.max(intent.confidence, sentiment.confidence)
    };
  }
}

// 游戏控制器集成
class AIGameController {
  constructor() {
    this.danmakuAgent = new DanmakuUnderstandingAgent();
    this.npcManager = new NPCManager();
    this.replyGenerator = new ReplyGenerator();
  }

  async handleDanmaku(danmaku) {
    const understanding = await this.danmakuAgent.process(danmaku.text);
    
    if (understanding.confidence > 0.8) {
      // 高置信度：执行对应游戏动作
      this.executeGameAction(understanding);
      
      // AI 角色回复
      const reply = await this.replyGenerator.generate({
        context: this.getGameContext(),
        sentiment: understanding.sentiment,
        npc_personality: this.npcManager.getCurrentNPCPersonality()
      });
      
      this.broadcastDanmakuReply(reply);
    }
  }
}
```
