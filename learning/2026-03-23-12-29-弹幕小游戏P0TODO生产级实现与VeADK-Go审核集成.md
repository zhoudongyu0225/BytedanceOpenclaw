// slg评论
func ChatMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    for _, msg := range messageList.MsgList {
        zlog.Logger.Info().Msgf("玩家评论 %v %v %v %v ", msg.Uid, msg.Name, msg.Content, ws.AccountId)
        if msg.Uid == "" {
            untils.TapErr(fmt.Sprintf("msg.Uid is 空 %v", msg))
            continue
        }
        pdb, _ := GetOpenVo(msg.Uid)
        if pdb == nil {
            untils.TapErr(fmt.Sprintf("pdb is nil %v", msg))
            continue
        }
        if msg.Uid != "" { pdb.OpenId = msg.Uid }
        if msg.Name != "" { pdb.NickName = msg.Name }
        if msg.HeadImg != "" { pdb.AvatarUrl = msg.HeadImg }

        if now-pdb.LastUpdateTime > 5 {
            currWeekRank, currWeekScore := model.GetGameWeekRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            currMonthRank, currMonthScore := model.GetGameMonthRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            pdb.Score = currWeekScore
            pdb.MonthScore = currMonthScore
            pdb.Rank = int32(currWeekRank)
            pdb.MonthRank = int32(currMonthRank)
            pdb.LastUpdateTime = now
        }

        // ===== P0 TODO:弹幕逻辑 =====
        ctx := context.Background()

        // 1. 敏感词过滤（DFA算法，毫秒级）
        sensitiveFilter := untils.GetSensitiveFilter()
        if sensitiveFilter.ContainsSensitiveWord(msg.Content) {
            zlog.Logger.Debug().Msgf("敏感词过滤: openId=%v content=%v", msg.Uid, msg.Content)
            ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
                TipMsg: "弹幕包含敏感词，请文明发言",
                Type:   pb.TipType_Normal,
            })
            continue
        }

        // 2. 弹幕冷却检查（Redis原子操作，5秒间隔）
        danmakuCoolKey := fmt.Sprintf("DanmakuCool.%v", msg.Uid)
        exists, err := model.GetRedisClient().Exists(ctx, danmakuCoolKey).Result()
        if err != nil {
            zlog.Logger.Error().Msgf("Redis冷却检查失败: %v", err)
        }
        if exists > 0 {
            zlog.Logger.Debug().Msgf("弹幕冷却中: openId=%v", msg.Uid)
            continue
        }

        // 3. 规则引擎匹配（DanmakuRule.json → action）
        action := matchDanmakuRule(msg.Content)
        if action == nil {
            // 无匹配规则，静默通过（普通弹幕）
            action = &DanmakuAction{
                ActionType: "normal",
                Score:      0,
                Broadcast:  false,
            }
        }

        // 4. 扣除用户积分（根据action类型）
        if action.Score > 0 && pdb.Score < action.Score {
            ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
                TipMsg: fmt.Sprintf("积分不足，当前%v，需要%v", pdb.Score, action.Score),
                Type:   pb.TipType_Normal,
            })
            continue
        }
        if action.Score > 0 {
            // 原子扣减积分
            newScore := atomicAddScore(pdb.OpenId, ws.AnchorV.PlatformId, -action.Score)
            pdb.Score = newScore
        }

        // 5. 执行游戏动作（士兵生成/技能触发等）
        if action.ActionType != "normal" {
            executeGameAction(ws, action, pdb)
        }

        // 6. 广播弹幕提示（带前缀标识 GAME_ACTION）
        broadcastMsg := fmt.Sprintf("GAME_ACTION|%v|%v|%v", msg.Name, msg.Content, action.ActionType)
        ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
            TipMsg: broadcastMsg,
            Type:   pb.TipType_RoomRoll,
        })

        // 7. 记录弹幕冷却（Redis TTL=5秒）
        model.GetRedisClient().Set(ctx, danmakuCoolKey, "1", 5*time.Second)
        // ===== P0 TODO END =====
    }
}

        currMonthRank, currMonthScore := model.GetGameMonthRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
        pdb.Score = currWeekScore
        pdb.MonthScore = currMonthScore
        pdb.Rank = int32(currWeekRank)

        pdb.LastUpdateTime = now

        ctx := context.Background()

        // 1. 获取礼物配置（Gift.json → GiftId映射）
        giftConfig := getGiftConfig(msg.GiftId)
        if giftConfig == nil {
            zlog.Logger.Warn().Msgf("礼物配置不存在 GiftId=%v", msg.GiftId)
            continue
        }

        // 2. 礼物价值校验（防止虚假数据）
        expectedTotal := int64(untils.StringToInt32(msg.GiftCount)) * giftConfig.UnitPrice
        actualTotal := untils.StringToInt32(msg.Total)
        if expectedTotal != actualTotal && giftConfig.UnitPrice > 0 {
            zlog.Logger.Warn().Msgf("礼物价值校验失败 GiftId=%v expected=%v actual=%v",
                msg.GiftId, expectedTotal, actualTotal)
            continue
        }

        // 3. 计算士兵数（根据GiftId和GiftCount）
        soldierNum := computeSoldierNum(giftConfig, msg.GiftCount)

        // 4. 计算救援与伤害值
        rescueNum := int64(0)
        injuryNum := int64(0)
        if len(giftConfig.RescueInjuryNum) >= 2 {
            rescueNum = int64(giftConfig.RescueInjuryNum[0]) * int64(untils.StringToInt32(msg.GiftCount))
            injuryNum = int64(giftConfig.RescueInjuryNum[1]) * int64(untils.StringToInt32(msg.GiftCount))
        }

        // 5. 原子增加用户积分
        newScore := atomicAddScore(pdb.OpenId, ws.AnchorV.PlatformId, actualTotal)
        pdb.Score = newScore

        // 6. 更新周榜+月榜
        model.UpdateGameWeekRank(pdb.OpenId, int64(actualTotal), ws.AnchorV.PlatformId)
        model.UpdateGameMonthRank(pdb.OpenId, int64(actualTotal), ws.AnchorV.PlatformId)

        // 7. 累计主播礼物价值
        ws.giftV += int64(untils.StringToInt32(msg.GiftCount)) * giftConfig.UnitPrice

        // 8. 广播礼物消息（GAME_GIFT前缀）
        broadcastMsg := fmt.Sprintf("GAME_GIFT|%v|%v|%v|%v", msg.Name, msg.GiftName, soldierNum, rescueNum)
        ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
            TipMsg: broadcastMsg,
            Type:   pb.TipType_RoomRoll,
        })

    }
}

func computeSoldierNum(cfg *GiftConfig, giftCount string) int64 {
    count := int64(untils.StringToInt32(giftCount))
    if count <= 0 {
        count = 1
    }
    if len(cfg.GetSoldierNum) == 0 {
        return 0
    }
    // GetSoldierNum[0]是基础值
    return int64(cfg.GetSoldierNum[0]) * count
}

func atomicAddScore(openId string, platformId int32, delta int64) int64 {
    ctx := context.Background()
    weekKey := fmt.Sprintf("PlayerScore.%v.%v.%v", openId, platformId, untils.GetWeekZeroTimestamp())
    newScore, err := model.GetRedisClient().IncrBy(ctx, weekKey, delta).Result()
    if err != nil {
        zlog.Logger.Error().Msgf("原子积分操作失败: %v", err)
        return 0
    }
    return newScore
}

func LikeMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    for _, msg := range messageList.MsgList {
        if msg.Uid == "" { continue }
        pdb, _ := GetOpenVo(msg.Uid)
        if pdb == nil { continue }

        if now-pdb.LastUpdateTime > 60 {
            currWeekRank, currWeekScore := model.GetGameWeekRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            currMonthRank, currMonthScore := model.GetGameMonthRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            pdb.Score = currWeekScore
            pdb.MonthScore = currMonthScore
            pdb.Rank = int32(currWeekRank)
            pdb.MonthRank = int32(currMonthRank)
            pdb.LastUpdateTime = now
        }

        // ===== P0 TODO:点赞逻辑 =====
        ctx := context.Background()
        likeCount := int64(untils.StringToInt32(msg.GetCount()))
        if likeCount <= 0 {
            likeCount = 1
        }

        // 1. 点赞冷却检查（防止刷屏，TTL=1秒）
        likeCoolKey := fmt.Sprintf("LikeCool.%v", msg.Uid)
        exists, err := model.GetRedisClient().Exists(ctx, likeCoolKey).Result()
        if err != nil {
            zlog.Logger.Error().Msgf("Redis点赞冷却检查失败: %v", err)
        }
        if exists > 0 {
            // 在冷却中，但仍然累计（点赞合并已在convergeMessage处理）
            continue
        }

        // 2. 累计点赞数（按日重置）
        dateKey := time.Now().Format("20060102")
        likeAccumKey := fmt.Sprintf("LikeAccum.%v.%v", msg.Uid, dateKey)
        totalLikes, err := model.GetRedisClient().IncrBy(ctx, likeAccumKey, likeCount).Result()
        if err != nil {
            zlog.Logger.Error().Msgf("Redis点赞累计失败: %v", err)
            totalLikes = likeCount
        }
        // 设置每日重置TTL（如果key是新建的）
        model.GetRedisClient().Expire(ctx, likeAccumKey, 24*time.Hour)

        // 3. 检查Buff触发阈值（Buff.json）
        triggered, buff := checkAndTriggerBuff(msg.Uid, totalLikes)
        if triggered && buff != nil {
            // 4. 触发Buff效果
            executeBuffEffect(ws, buff, pdb)

            // 5. 记录Buff触发（防止重复触发）
            buffTriggerKey := fmt.Sprintf("BuffTriggered.%v.%v.%v", msg.Uid, buff.Id, dateKey)
            model.GetRedisClient().Set(ctx, buffTriggerKey, "1", 24*time.Hour)

            // 6. 广播点赞Buff触发（GAME_BUFF前缀）
            broadcastMsg := fmt.Sprintf("GAME_BUFF|%v|%v|%v|%v", msg.Name, totalLikes, buff.BuffType, buff.Int_Param)
            ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
                TipMsg: broadcastMsg,
                Type:   pb.TipType_RoomRoll,
            })
        }

        // 7. 记录点赞冷却
        model.GetRedisClient().Set(ctx, likeCoolKey, "1", 1*time.Second)
        // ===== P0 TODO END =====
    }
}

// checkAndTriggerBuff 检查并触发Buff
func checkAndTriggerBuff(openId string, totalLikes int64) (bool, *BuffConfig) {
    buffs := getAllBuffConfigs() // 从Buff.json加载
    dateKey := time.Now().Format("20060102")

    for _, buff := range buffs {
        if totalLikes >= int64(buff.Threshold) {
            // 检查是否已触发
            buffTriggerKey := fmt.Sprintf("BuffTriggered.%v.%v.%v", openId, buff.Id, dateKey)
            exists, _ := model.GetRedisClient().Exists(context.Background(), buffTriggerKey).Result()
            if exists == 0 {
                return true, buff
            }
        }
    }
    return false, nil
}

// 弹幕内容审核Callback
func danmakuAuditCallback(ctx agent.CallbackContext, req *model.LLMRequest) (*model.LLMResponse, error) {
    // 提取用户弹幕内容
    for _, content := range req.Contents {
        for _, part := range content.Parts {
            if text := part.Text; text != "" {
                // 调用DFA预检
                filter := GetSensitiveFilter()
                if filter.ContainsSensitiveWord(text) {
                    // 直接返回拒绝响应，跳过LLM调用
                    return &model.LLMResponse{
                        Content: &genai.Content{
                            Parts: []*genai.Part{{Text: `{"action":"reject","reason":"sensitive_word_detected"}`}},
                            Role:  "model",
                        },
                    }, nil
                }
            }
        }
    }
    return nil, nil // 继续正常流程
}

// AfterModelCallback 实现（审核结果解析）
func auditResultCallback(ctx agent.CallbackContext, resp *model.LLMResponse, respErr error) (*model.LLMResponse, error) {
    if respErr != nil {
        return nil, respErr
    }
    if resp == nil || resp.Content == nil || len(resp.Content.Parts) == 0 {
        return nil, nil
    }

    // 解析JSON审核结果
    originalText := resp.Content.Parts[0].Text
    var auditResult AuditResult
    if err := json.Unmarshal([]byte(originalText), &auditResult); err != nil {
        // JSON解析失败，降级为通过
        return nil, nil
    }

    if auditResult.Action == "reject" {
        // 记录审核拒绝日志
        zlog.Logger.Warn().Msgf("AI审核拒绝: action=%v reason=%v", auditResult.Action, auditResult.Reason)
        return &model.LLMResponse{
            Content: &genai.Content{
                Parts: []*genai.Part{{Text: fmt.Sprintf(`{"action":"reject","reason":"%v"}`, auditResult.Reason)}},
                Role:  "model",
            },
        }, nil
    }

    return nil, nil
}

// 创建审核Agent
func newAuditAgent() (agent.Agent, error) {
    return veagent.New(&veagent.Config{
        Config: llmagent.Config{
            Name:        "barrage_audit",
            Description: "弹幕内容安全审核Agent",
            Instruction: `你是一个严格的弹幕内容安全审核系统。
收到弹幕内容后，判断其安全性：
- 如果弹幕包含色情低俗内容、政治敏感内容、人身攻击、虚假信息，返回 {"action":"reject","reason":"<具体原因>"}
- 如果弹幕内容安全正常，返回 {"action":"pass","reason":""}
- 如果弹幕内容边界模糊但可能是恶意的，返回 {"action":"review","reason":"<建议人工复核的原因>"}

注意：只返回JSON格式，不要有其他文字。`,
            BeforeModelCallbacks: []llmagent.BeforeModelCallback{danmakuAuditCallback},
            AfterModelCallbacks:   []llmagent.AfterModelCallback{auditResultCallback},
        },
        ModelName: "doubao-pro-32k",
    })
}
