```go
        ScoreBonus:  2,
    },

// 规则引擎
type DanmakuEngine struct {
    rules      []DanmakuRule
    cooldowns  map[string]int64 // "openid_action" -> last_trigger_timestamp
    mu         sync.RWMutex
}

func NewDanmakuEngine() *DanmakuEngine {
    return &DanmakuEngine{
        rules:     danmakuRules,
        cooldowns: make(map[string]int64),
    }
}

func (e *DanmakuEngine) Process(content string, openId string) *GameCommand {
    e.mu.Lock()
    defer e.mu.Unlock()
    
    lower := strings.ToLower(content)
    now := time.Now().Unix()
    
    for _, rule := range e.rules {
        for _, kw := range rule.Keywords {
            if strings.Contains(lower, kw) {
                // 检查冷却
                key := openId + "_" + fmt.Sprintf("%d", rule.ActionType)
                if lastTime, ok := e.cooldowns[key]; ok {
                    if now-lastTime < int64(rule.CooldownSec) {
                        return nil // 冷却中，跳过
                    }
                }
                
                // 检查敏感词（复用现有模块）
                if untils.SensitiveCheck(content) {
                    zlog.Logger.Warn().Msgf("弹幕含敏感词，跳过: %v", content)
                    return nil
                }
                
                // 触发成功
                e.cooldowns[key] = now
                
                // 增加玩家分数
                model.AddPlayerScore(openId, rule.ScoreBonus)
                
                return &GameCommand{
                    ActionType:  rule.ActionType,
                    ActionParam: rule.ActionParam,
                    OpenId:      openId,
                    Timestamp:   now,
                }
            }
        }
    }
    return nil
}
```

```go
            msg.Name, msg.Content, cmd.ActionType, cmd.ActionParam)
    }
    
    // 批量广播给UE4客户端
    if len(commands) > 0 {

            Commands: commands,
```

```go
// 在 control_barrage.go 的 GiftMessage 函数中填充

func GiftMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    
    var commands []*pb.DanmakuCommand
    
    for _, msg := range messageList.MsgList {
        // 已有逻辑：玩家数据更新
        // ...（原代码保持不变）...
        
        // ===== 新增：礼物逻辑 =====
        giftId := msg.GiftId    // 来自 pb.Message.GiftId
        giftNum := 1
        if msg.GiftCount != "" {
            giftNum, _ = strconv.Atoi(msg.GiftCount)
        }
        totalValue := 0
        if msg.Total != "" {
            totalValue, _ = strconv.Atoi(msg.Total)
        }
        
        // 查找礼物映射
        mapping := findGiftMapping(giftId)
        if mapping == nil {
            zlog.Logger.Warn().Msgf("未配置的礼物ID: %s", giftId)
            continue
        }
        
        // 检查是否满足最小价值
        if totalValue < mapping.MinValue {
            continue
        }
        
        // 检查冷却（按礼物类型）
        key := msg.Uid + "_gift_" + giftId
        if isInCooldown(key, 10) { // 10秒礼物冷却
            continue
        }
        
        // 触发游戏命令
        for i := 0; i < giftNum; i++ {
            cmd := &pb.DanmakuCommand{
                OpenId:     msg.Uid,
                ActionType: mapping.GameAction,
                ActionParam: mapping.ActionParam,
                Timestamp:  now,
                Nickname:  msg.Name,
            }
            commands = append(commands, cmd)
        }
        
        // 增加玩家分数（礼物价值的10%作为积分）
        scoreBonus := totalValue / 10
        model.AddPlayerScore(msg.Uid, int32(scoreBonus))
        
        // 更新礼物统计
        model.UpdateGiftStats(msg.Uid, giftId, giftNum, totalValue)
        
        zlog.Logger.Info().Msgf("礼物触发 [%s] %s x%d -> action:%d param:%s",
            msg.Name, mapping.GiftName, giftNum, mapping.GameAction, mapping.ActionParam)
    }
    
    // 批量广播
    if len(commands) > 0 {
        broadcast := &pb.DanmakuCommandBroadcast{
            Commands: commands,
            RoomId:   ws.AnchorV.RoomId,
        }
        ws.WsSend(COMMAND_GIFT_BROADCAST, broadcast)
    }
}
```

## 七、行业案例参考

### 7.1 弹幕游戏核心玩法设计

**典型弹幕游戏玩法模式**：

1. **弹幕控制型**：
   - 玩家发"左"→ 游戏角色左移
   - 玩家发"攻击"→ 释放技能
   - 适合：动作类、休闲类

2. **投票决策型**：
   - 玩家通过弹幕投票决定游戏走向
   - 例如："龙妈VS狮家"选择阵营
   - 适合：策略类、剧情类

3. **资源贡献型**：
   - 礼物/弹幕累积资源
   - 资源达到阈值触发游戏事件
   - 适合：塔防类、放置类

4. **组合型（推荐）**：
   - 低成本弹幕："左/右/攻击"→ 基础操作
   - 付费礼物：直接召唤高级恐龙/BOSS
   - 点赞：全体BUFF
