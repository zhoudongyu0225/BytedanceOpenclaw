```go
package main

import (
    "context"
    "fmt"
    "os"

    _ "github.com/volcengine/veadk-go/agent"
    veagent "github.com/volcengine/veadk-go/agent/llmagent"
    "github.com/volcengine/veadk-go/log"
    "google.golang.org/adk/agent"
    "google.golang.org/adk/cmd/launcher"
    "google.golang.org/adk/cmd/launcher/full"
    "google.golang.org/adk/session"
)

func main() {
    ctx := context.Background()
    veAgent, err := veagent.New(&veagent.Config{
        ModelExtraConfig: map[string]any{
            "extra_body": map[string]any{
                "thinking": map[string]string{
                    "type": "disabled",  // 关闭思维链，减少延迟
                },
            },
        },
    })
    if err != nil {
        log.Errorf("NewVeAgent failed: %v", err)
        return
    }

    config := &launcher.Config{
        AgentLoader:     agent.NewSingleLoader(veAgent),
        SessionService: session.InMemoryService(),
    }

    l := full.NewLauncher()
    if err = l.Execute(ctx, config, os.Args[1:]); err != nil {
        log.Fatalf("Run failed: %v\n\n%s", err, l.CommandLineSyntax())
    }
}
```

**运行模式**:
- CLI: `go run agent.go`
- Web UI: `go run agent.go web api webui`
- 带超时: `go run agent.go web -read-timeout 3m -write-timeout 3m api webui`
