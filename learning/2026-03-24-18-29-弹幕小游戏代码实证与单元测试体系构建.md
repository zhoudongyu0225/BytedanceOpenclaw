func (sf *SensitiveFilter) ContainsSensitiveWord(text string) bool {
    return sf.filter.Filter(text) != text  // DFA算法, O(n)复杂度
}

func (sf *SensitiveFilter) ReplaceSensitiveWords(text string, replaceWith rune) string {
    return sf.filter.Replace(text, replaceWith)
}

func ChatMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)

    for _, msg := range messageList.MsgList {
        // 1. 基础校验
        if msg.Uid == "" || msg.Content == "" {
            continue
        }

        // 2. 获取玩家数据
        pdb, _ := GetOpenVo(msg.Uid)
        if pdb == nil {
            continue
        }

        // 3. 敏感词过滤(使用已实现的过滤器)
        filter := untils.GetSensitiveFilter()
        if filter.ContainsSensitiveWord(msg.Content) {
            // 拦截敏感词,发送提示(可选项,避免刷屏)
            ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
                Type:    0, // Normal
                Content: "消息含有敏感词",
            })
            continue
        }

        // 4. 弹幕冷却检查(Redis原子操作,5秒)
        coolKey := fmt.Sprintf("DanmakuCool:%s:%s", ws.AccountId, msg.Uid)
        redisCli := model.GetRedisClient()
        exists, _ := redisCli.Exists(context.Background(), coolKey).Result()
        if exists > 0 {
            // 冷却中,静默跳过(不提示,避免刷屏)
            continue
        }

        // 5. 弹幕规则匹配(可扩展,预留接口)
        // rule := model.MatchDanmakuRule(msg.Content)
        // if rule != nil { ... }

        // 6. 积分扣除(如果规则要求消费积分)
        // model.DeductScore(msg.Uid, rule.ScoreCost)

        // 7. 设置冷却(原子操作)
        redisCli.SetNX(context.Background(), coolKey, "1", 5*time.Second)

        // 8. 刷新玩家积分/排行(复用已有逻辑,减少Redis调用)
        if now-pdb.LastUpdateTime > 5 {
            currWeekRank, currWeekScore := model.GetGameWeekRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            currMonthRank, currMonthScore := model.GetGameMonthRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            pdb.Score = currWeekScore
            pdb.MonthScore = currMonthScore
            pdb.Rank = int32(currWeekRank)
            pdb.MonthRank = int32(currMonthRank)
            pdb.LastUpdateTime = now
        }

        // 9. 广播游戏动作(使用TipsNotify, GameAction类型)
        ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
            Type:    4, // GameAction
            Content: buildGameActionPayload(msg.Uid, msg.Name, msg.Content),
        })

        zlog.Logger.Debug().Msgf("弹幕处理完成 uid=%s content=%s", msg.Uid, msg.Content)
    }
}

    return fmt.Sprintf(`{"uid":"%s","name":"%s","content":"%s","action":"chat"}`,

func LikeMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    if messageList == nil || len(messageList.MsgList) == 0 {
        return
    }

    now := time.Now().Unix()
    redisCli := model.GetRedisClient()

    for _, msg := range messageList.MsgList {
        if msg.Uid == "" {
            continue
        }

        pdb, _ := GetOpenVo(msg.Uid)
        if pdb == nil {
            continue
        }

        // 1. 点赞数解析
        likeCount := StirngToInt32(msg.Count)
        if likeCount <= 0 {
            continue
        }

        // 2. 点赞冷却检查(5秒)
        coolKey := fmt.Sprintf("LikeCool:%s:%s", ws.AccountId, msg.Uid)
        exists, _ := redisCli.Exists(context.Background(), coolKey).Result()
        if exists > 0 {
            continue
        }

        // 3. 累计点赞(Redis原子累加,60秒过期)
        accumKey := fmt.Sprintf("LikeAccum:%s:%s", ws.AccountId, msg.Uid)
        newAccum, _ := redisCli.IncrBy(context.Background(), accumKey, int64(likeCount)).Result()
        redisCli.Expire(context.Background(), accumKey, 60*time.Second)

        // 4. 设置冷却(5秒)
        redisCli.SetNX(context.Background(), coolKey, "1", 5*time.Second)

        // 5. Buff阈值检查(10/20/50/100)
        buffLevel := getBuffLevel(newAccum)
        if buffLevel > 0 {
            // 触发Buff广播
            ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
                Type:    6, // GameBuff
                Content: buildGameBuffPayload(msg.Uid, msg.Name, buffLevel, newAccum),
            })
            zlog.Logger.Debug().Msgf("点赞Buff触发 uid=%s level=%d accum=%d",
                msg.Uid, buffLevel, newAccum)
        }

        // 6. 刷新玩家排行(点赞不频繁,复用60秒逻辑)
        if now-pdb.LastUpdateTime > 60 {
            currWeekRank, currWeekScore := model.GetGameWeekRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
            pdb.Score = currWeekScore
            pdb.Rank = int32(currWeekRank)
            pdb.LastUpdateTime = now
        }
    }
}

// getBuffLevel 根据累计点赞数返回Buff等级
// 阈值: 10→1级, 20→2级, 50→3级, 100→4级
func getBuffLevel(accumLikes int64) int32 {
    switch {
    case accumLikes >= 100:
        return 4
    case accumLikes >= 50:
        return 3
    case accumLikes >= 20:
        return 2
    case accumLikes >= 10:
        return 1
    default:
        return 0
    }
}

func buildGameBuffPayload(uid, name string, level int32, accum int64) string {
    return fmt.Sprintf(`{"uid":"%s","name":"%s","level":%d,"accum":%d,"action":"buff"}`,
        uid, name, level, accum)
}
