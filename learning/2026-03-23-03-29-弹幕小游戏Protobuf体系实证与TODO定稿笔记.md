**关键发现**：
- `count` 字段用于**点赞数量**（LikeMessage），需用 `strconv.ParseInt(msg.Count, 10, 32)` 解析
- `giftId` + `giftCount` + `giftName` + `total` 用于**礼物消息**
- `content` 用于**弹幕文本**
- 三类消息共用同一个 `Message` 结构，通过 `type` 字段区分

```go
  }
  if msg.Name != "" {
   pdb.NickName = msg.Name
  }
  if msg.HeadImg != "" {
   pdb.AvatarUrl = msg.HeadImg
  }

  currWeekRank, currWeekScore := model.GetGameWeekRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
  currMonthRank, currMonthScore := model.GetGameMonthRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
  pdb.Score = currWeekScore
  pdb.MonthScore = currMonthScore
  pdb.Rank = int32(currWeekRank)
  pdb.MonthRank = int32(currMonthRank)
  pdb.LastUpdateTime = now

  // ========== 【TODO:礼物逻辑】完整实现 ==========
  // 1. 礼物配置查询（从缓存或配置文件）
  giftConfig := GetGiftConfig(msg.GiftId)
  if giftConfig == nil {
   zlog.Logger.Warn().Msgf("未知礼物ID: %s", msg.GiftId)
   continue
  }

  // 2. 计算士兵数量和资源
  // 基础士兵 = giftCount * giftConfig.SoldierRatio
  // 额外士兵（根据礼物价值阶梯）
  soldierNum := computeSoldierNum(giftConfig, msg.GiftCount)
  rescueNum := computeRescueNum(giftConfig, msg.GiftCount)

  // 3. 价值校验（防止刷礼物）
  totalValue := untils.StringToInt64(msg.Total)
  expectedValue := int64(giftConfig.Value) * untils.StringToInt64(msg.GiftCount)
  if totalValue < expectedValue*80/100 {
    zlog.Logger.Warn().Msgf("礼物价值异常: giftId=%s total=%s expected=%d",
     msg.GiftId, msg.Total, expectedValue)
    continue
  }

  // 4. 积分/贡献值增加
  contributeScore := int32(giftConfig.Value) * untils.StringToInt32(msg.GiftCount)
  pdb.Score += contributeScore
  pdb.WeekScore += contributeScore

  // 5. 更新玩家数据
  pdb.TotalGiftValue += totalValue
  pdb.GiftCount++

  // 6. 构造游戏指令并广播
  ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
   TipMsg: fmt.Sprintf("GAME_GIFT|%s|%s|%s|%d|%d",
    msg.Uid, msg.Name, giftConfig.Name, soldierNum, rescueNum),
   Type: pb.TipType_RoomRoll,
  })

  // 7. 更新排行榜（异步）
  go func() {
   model.UpdateGameWeekRank(pdb.OpenId, ws.AnchorV.PlatformId, contributeScore)
  }()

  // 8. 记录礼物日志
  zlog.Logger.Info().Msgf("礼物触发: uid=%s gift=%s count=%d soldiers=%d rescue=%d",
   msg.Uid, giftConfig.Name, msg.GiftCount, soldierNum, rescueNum)
 }
}
```

```go
func LikeMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
 now := time.Now().Unix()
 msgJoin(messageList, ws)
 for _, msg := range messageList.MsgList {
  if msg.Uid == "" {
   untils.TapErr(fmt.Sprintf("平台发的 uid is nil %v", msg))
   continue
  }
  pdb, _ := GetOpenVo(msg.Uid)
  if pdb == nil {
   untils.TapErr(fmt.Sprintf("pdb is nil %v", msg))
   continue
  }
  if now-pdb.LastUpdateTime > 60 {
   currWeekRank, currWeekScore := model.GetGameWeekRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
   currMonthRank, currMonthScore := model.GetGameMonthRankByRankVo(pdb.OpenId, ws.AnchorV.PlatformId)
   pdb.Score = currWeekScore
   pdb.MonthScore = currMonthScore
   pdb.Rank = int32(currWeekRank)
   pdb.MonthRank = int32(currMonthRank)
   pdb.LastUpdateTime = now
  }

  // ========== 【TODO:点赞逻辑】完整实现 ==========
  // 1. 获取点赞数量（msg.Count 字段）
  likeCount := untils.StringToInt32(msg.Count)
  if likeCount <= 0 {
   likeCount = 1 // 默认1个赞
  }

  // 2. 累计点赞数（Redis原子操作）
  likeKey := fmt.Sprintf("like:累计:%s:%s", ws.AnchorV.AccountId, msg.Uid)
  totalLikes := RedisIncrBy(likeKey, int64(likeCount))

  // 3. 获取点赞Buff阈值配置
  buffConfigs := GetLikeBuffConfigs() // e.g., {10, 20, 50, 100}
  triggeredBuff := getTriggeredBuff(buffConfigs, totalLikes)

  // 4. Buff触发判断（只在达到阈值时触发）
  if triggeredBuff != nil {
   // 检查是否已触发过该档位
   triggeredKey := fmt.Sprintf("like:triggered:%s:%s:%d",
    ws.AnchorV.AccountId, msg.Uid, triggeredBuff.Threshold)
   if !RedisBoolCheck(triggeredKey) {
    // 5. 触发Buff效果
    ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
     TipMsg: fmt.Sprintf("GAME_BUFF|%s|%s|%s|%d",
      msg.Uid, msg.Name, triggeredBuff.BuffType, triggeredBuff.BuffValue),
     Type: pb.TipType_RoomRoll,
    })
    // 标记已触发
    RedisSetWithExpire(triggeredKey, "1", 86400) // 24小时不重复触发
   }
  }

  // 6. 定期更新数据库（每累计100赞或每60秒）
  if totalLikes%100 == 0 || now-pdb.LastUpdateTime > 60 {
   go func() {
    model.UpdatePlayerLikeCount(pdb.OpenId, totalLikes)
   }()
  }
 }
}
```

### 3.2 规则引擎核心实现

```go
type DanmakuRule struct {
 RuleId      string
 Keywords    []string
 Action      string
 ActionParam map[string]interface{}
 ScoreCost   int32
 CooldownSec int64
 Platforms   []string
 Priority    int
}

type LikeBuffConfig struct {
 Threshold  int32
 BuffType   string
 BuffValue  float64
 DurationSec int64
}

// MatchDanmakuRule - 规则匹配（支持多关键词 + 优先级）
func MatchDanmakuRule(content string) *DanmakuRule {
 content = strings.TrimSpace(content)
 for _, rule := range globalRuleCache.SortedByPriority() {
  // 平台过滤
  if !contains(rule.Platforms, currentPlatform) {
   continue
  }
  // 关键词匹配
  for _, kw := range rule.Keywords {
   if strings.Contains(content, kw) {
    return rule
   }
  }
 }
 return nil
}

// getTriggeredBuff - 获取触发阈值对应的Buff
func getTriggeredBuff(configs []LikeBuffConfig, totalLikes int64) *LikeBuffConfig {
 var triggered *LikeBuffConfig
 for i := range configs {
  if configs[i].Threshold <= totalLikes {
   if triggered == nil || configs[i].Threshold > triggered.Threshold {
    triggered = &configs[i]
   }
  }
 }
 return triggered
}
```

### 4.1 computeSoldierNum - 士兵数量计算

```go
// GiftConfig - 礼物配置结构
type GiftConfig struct {
 GiftId        string
 Name          string
 Value         int32   // 单个礼物价值
 SoldierRatio  int32   // 每个礼物产生的士兵数
 ExtraRatios   map[int32]int32 // 阶梯奖励：数量→额外士兵比例
}

// computeSoldierNum - 根据礼物数量计算士兵数
func computeSoldierNum(cfg *GiftConfig, giftCountStr string) int32 {
 count := untils.StringToInt32(giftCountStr)
 if count <= 0 {
  count = 1
 }
 baseSoldiers := count * cfg.SoldierRatio

 // 阶梯额外奖励
 extraRatio := int32(0)
 for threshold, ratio := range cfg.ExtraRatios {
  if count >= threshold {
   extraRatio = ratio
  }
 }
 extraSoldiers := (count * ratio) / 100 // 百分比额外士兵

 return baseSoldiers + extraSoldiers
}

// computeRescueNum - 计算治疗/救援数量
func computeRescueNum(cfg *GiftConfig, giftCountStr string) int32 {
 count := untils.StringToInt32(giftCountStr)
 if count <= 0 {
  count = 1
 }
 // 治疗量 = 礼物数量 * 基础治疗系数
 return count * cfg.HealRatio
}
```

### 4.2 Redis 原子操作辅助

```go
// RedisBoolCheck - 检查key是否存在
func RedisBoolCheck(key string) (bool, error) {
 result, err := redis.Bool(redisClient.Do("EXISTS", key))
 return result, err
}

// RedisIncrBy - 原子递增
func RedisIncrBy(key string, delta int64) int64 {
 result, err := redis.Int64(redisClient.Do("INCRBY", key, delta))
 if err != nil {
  zlog.Logger.Error().Msgf("RedisIncrBy error: %v", err)
  return 0
 }
 return result
}

// RedisSetWithExpire - 设置带过期时间的key
func RedisSetWithExpire(key string, value string, expireSec int64) {
 redisClient.Do("SETEX", key, expireSec, value)
}
```
