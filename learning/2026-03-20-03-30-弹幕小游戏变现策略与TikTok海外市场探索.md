```javascript
class AdManager {
    constructor() {
        this.rewardedVideo = null;
        this.initRewardedVideo();
    }

    initRewardedVideo() {
        if (typeof tt != 'undefined' && tt.createRewardedVideoAd) {
            this.rewardedVideo = tt.createRewardedVideoAd({
                adUnitId: 'your_ad_unit_id'
            });
            
            this.rewardedVideo.onLoad(() => {
                console.log('激励视频加载成功');
            });
            
            this.rewardedVideo.onError((err) => {
                console.log('激励视频加载失败', err);
            });
        }
    }

    async showRewardedVideo(rewardType) {
        try {
            await this.rewardedVideo.show();
            
            this.rewardedVideo.onClose((res) => {
                if (res.isEnded) {
                    // 用户完整观看，发放奖励
                    this.grantReward(rewardType);
                } else {
                    // 用户中途关闭，轻度奖励或不奖励
                    this.grantPartialReward(rewardType);
                }
            });
        } catch (err) {
            console.log('激励视频展示失败', err);
            // 降级：直接发放奖励
            this.grantReward(rewardType);
        }
    }

    grantReward(rewardType) {
        switch (rewardType) {
            case 'shield':
                // 获得护盾道具
                player.addItem('shield', 1);
                break;
            case 'speed_boost':
                // 获得加速道具
                player.addItem('speed_boost', 1);
                break;
            case 'revive':
                // 复活
                game.revive();
                break;
        }
    }
}
```

**接入方式：**
1. 通过 TikTok LIVE Studio（OBS 插件）



### 3.3 合规与审核要点

### 5.2 支付回调安全

```javascript
// 支付回调验签示例（抖音）
const crypto = require('crypto');

class PaymentCallbackHandler {
    verifySignature(params, appSecret) {
        const { sign, order_no, ...other } = params;
        
        // 排除 sign 参数，按字典序排序所有参数
        const sortedKeys = Object.keys(other).sort();
        let signString = '';
        
        for (const key of sortedKeys) {
            if (other[key] !== '' && other[key] !== null && other[key] !== undefined) {
                signString += `${key}=${other[key]}&`;
            }
        }
        signString += `appsecret=${appSecret}`;
        
        const expectedSign = crypto
            .createHmac('sha256', appSecret)
            .update(signString)
            .digest('hex');
        
        return sign === expectedSign;
    }

    async handleCallback(params) {
        if (!this.verifySignature(params, this.appSecret)) {
            throw new Error('签名验证失败');
        }

        // 幂等处理
        const order = await this.getOrderByOrderNo(params.order_no);
        if (order.status === 'paid') {
            return { success: true, message: '订单已处理' };
        }

        // 发放道具
        await this.deliverItem(order.user_id, order.product_id);
        
        // 更新订单状态
        await this.updateOrderStatus(order.id, 'paid');
        
        return { success: true };
    }
}
```
