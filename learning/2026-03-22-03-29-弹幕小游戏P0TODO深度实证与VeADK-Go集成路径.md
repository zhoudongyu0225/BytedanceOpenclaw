package control

import (
    "dmGameServer/generateConfig"
    "dmGameServer/zlog"
    "sync"
    "time"
)

// DanmakuRule 弹幕规则
type DanmakuRule struct {
    Keyword      string
    Action      string
    Score       int
    CooldownSec int
}

// DanmakuRuleEngine 规则引擎
type DanmakuRuleEngine struct {
    rules        []DanmakuRule
    cooldownMap  sync.Map // openId→lastTriggerTime
}

// 全局单例
var ruleEngine *DanmakuRuleEngine

func InitDanmakuRuleEngine() {
    ruleEngine = &DanmakuRuleEngine{
        rules: loadRules(),
    }
}

func GetDanmakuRuleEngine() *DanmakuRuleEngine {
    if ruleEngine == nil {
        InitDanmakuRuleEngine()
    }
    return ruleEngine
}

// Match 匹配弹幕内容，返回动作、积分、是否匹配
func (e *DanmakuRuleEngine) Match(content string) (action string, score int, matched bool) {
    for _, rule := range e.rules {
        if containsKeyword(content, rule.Keyword) {
            return rule.Action, rule.Score, true
        }
    }
    return "", 0, false
}

// CheckCooldown 检查玩家冷却
func (e *DanmakuRuleEngine) CheckCooldown(openId, action string) bool {
    key := openId + ":" + action
    if lastTimeVal, ok := e.cooldownMap.Load(key); ok {
        lastTime := lastTimeVal.(time.Time)
        // 查找对应规则的冷却时间
        for _, rule := range e.rules {
            if rule.Action == action {
                if time.Since(lastTime) < time.Duration(rule.CooldownSec)*time.Second {
                    return false
                }
            }
        }
    }
    e.cooldownMap.Store(key, time.Now())
    return true
}

func containsKeyword(content, keyword string) bool {
    // 简单实现：按字节匹配，支持中英文
    for i := 0; i <= len(content)-len(keyword); i++ {
        if content[i:i+len(keyword)] == keyword {
            return true
        }
    }
    return false
}

func loadRules() []DanmakuRule {
    // 硬编码规则（后续可扩展为从DanmakuRule.json加载）
    return []DanmakuRule{
        {"攻击", "attack", 10, 3},
        {"进攻", "attack", 10, 3},
        {"防守", "defend", 10, 3},
        {"治疗", "heal", 15, 5},
        {"大招", "ultimate", 50, 30},
    }
}

            pdb.Rank = int32(currWeekRank)

        // 1. 敏感词过滤
        sf := untils.GetSensitiveFilter()
        if sf.ContainsSensitiveWord(msg.Content) {
            zlog.Logger.Info().Msgf("弹幕过滤 [%s] %s", msg.Uid, msg.Content)
            msg.Content = sf.ReplaceSensitiveWords(msg.Content, '*')
        }
        
        // 2. 关键词规则引擎匹配
        engine := GetDanmakuRuleEngine()
        if action, score, matched := engine.Match(msg.Content); matched {

                    TipMsg: fmt.Sprintf("[%s]: %s", msg.Name, msg.Content),

                continue

func computeSoldierNum(cfg *config.GiftConfig, count int32) int32 {
    if cfg == nil || len(cfg.GiftNum) == 0 {
        return 0
    }
    // 遍历档位，找最接近的
    for i := len(cfg.GiftNum) - 1; i >= 0; i-- {
        if cfg.GiftNum[i] != -1 && count >= cfg.GiftNum[i] {
            if i < len(cfg.GetSoldierNum) {
                return cfg.GetSoldierNum[i]
            }
        }
    }
    // -1表示无档位限制，任何数量都获得对应士兵数
    if len(cfg.GiftNum) > 0 && cfg.GiftNum[0] == -1 {
        return cfg.GetSoldierNum[0]
    }
    return 0
}
