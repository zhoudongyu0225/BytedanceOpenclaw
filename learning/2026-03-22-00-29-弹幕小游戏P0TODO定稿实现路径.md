```go
func ChatMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    
    for _, msg := range messageList.MsgList {
        if msg.Uid == "" || pdb == nil {
            continue
        }
        
        // 已有代码: pdb数据更新（5秒间隔查Redis排行）
        // ========== 弹幕逻辑（TODO #1 实现）==========
        
        // 1. 敏感词过滤（已就绪: untils/sensitive.go）
        sf := untils.GetSensitiveFilter()
        if sf.ContainsSensitiveWord(msg.Content) {
            zlog.Logger.Info().Msgf("弹幕过滤 [%s] %s", msg.Uid, msg.Content)
            // 替换后仍显示，但不触发游戏动作
            msg.Content = sf.ReplaceSensitiveWords(msg.Content, '*')
        }
        
        // 2. 关键词规则引擎（新增: danmakuRuleEngine）
        // 配置文件: generateJsons/DanmakuRule.json（需新建）
        // 映射: 关键词 → 动作类型 → 游戏效果
        if action, score, matched := danmakuRuleEngine.Match(msg.Content); matched {
            // 3. 冷却检查（玩家级别冷却，防止刷屏）
            if !danmakuRuleEngine.CheckCooldown(msg.Uid, action) {
                continue
            }
            
            // 4. 积分更新
            pdb.Score += score
            model.UpdateGameWeekRank(pdb.OpenId, pdb.Score, ws.AnchorV.PlatformId)
            
            // 5. 触发游戏动作（→ UE4客户端）
            // 复用 TipType_Normal = 0，前缀协议化
            ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
                TipMsg: fmt.Sprintf("GAME_ACTION|%s|%s|%s|%d", 
                    action, msg.Uid, msg.Name, score),
                Type: pb.TipType_Normal,
            })
        }
        
        // 6. 弹幕展示（始终广播，用于UI显示）
        ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
            TipMsg: fmt.Sprintf("[%s]: %s", msg.Name, msg.Content),
            Type:   pb.TipType_RoomRoll, // 2=直播间滚条
        })
    }
}
```

**computeSoldierNum 实现**（基于 Gift.json 配置）:
```go
func computeSoldierNum(cfg *config.GiftConfig, count int32) int32 {
    if cfg == nil || len(cfg.GiftNum) == 0 {
        return 0
    }
    // 找最接近的档位
    for i := len(cfg.GiftNum) - 1; i >= 0; i-- {
        if cfg.GiftNum[i] != -1 && count >= cfg.GiftNum[i] {
            if i < len(cfg.GetSoldierNum) {
                return cfg.GetSoldierNum[i]
            }
        }
    }
    // -1 表示无档位限制
    if len(cfg.GiftNum) > 0 && cfg.GiftNum[0] == -1 {
        return cfg.GetSoldierNum[0]
    }
    return 0
}
```

```go
func LikeMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    
    for _, msg := range messageList.MsgList {
        if msg.Uid == "" || pdb == nil {
            continue
        }
        
        // 已有代码: 60秒间隔更新排行
        // ========== 点赞逻辑（TODO #3 实现）==========
        
        // 1. 解析点赞数
        likeCount, _ := strconv.ParseInt(msg.Count, 10, 64)
        if likeCount <= 0 {
            continue
        }
        
        // 2. 累计点赞
        pdb.LikeCount += likeCount
        
        // 3. 积分小额奖励
        scoreReward := likeCount * 1  // 1积分/点赞
        pdb.Score += scoreReward
        model.UpdateGameWeekRank(pdb.OpenId, pdb.Score, ws.AnchorV.PlatformId)
        
        // 4. 阈值Buff触发检查（Buff.json）
        // BuffType=1: 点赞阈值Buff
        // Int_Param: 触发阈值（10/20/50/100）
        // IntArray_Param: Buff参数
        for _, buffCfg := range config.GetBuffList() {
            if buffCfg.BuffType == 1 { // 点赞Buff
                if pdb.LikeCount >= int64(buffCfg.Int_Param) {
                    ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
                        TipMsg: fmt.Sprintf("GAME_BUFF|%s|%s|%d|%v",
                            msg.Uid, msg.Name, buffCfg.Id, buffCfg.IntArray_Param),
                        Type: pb.TipType_Normal,
                    })
                }
            }
        }
        
        // 5. 点赞特效（阈值触发，避免刷屏）
        // 每10/50/100次触发一次展示
        if pdb.LikeCount == 10 || pdb.LikeCount == 50 || 
           pdb.LikeCount == 100 || pdb.LikeCount%100 == 0 {
            ws.WsSend(COMMAND_TIPS_S, &pb.TipsNotify{
                TipMsg: fmt.Sprintf("[%s] 点赞突破 %d！", msg.Name, pdb.LikeCount),
                Type:   pb.TipType_RoomRoll,
            })
        }
    }
}
```
