```go
// DanmakuWorkerPool 弹幕Worker池
type DanmakuWorkerPool struct {
    jobChan  chan DanmakuJob
    wg       sync.WaitGroup
    quit     chan struct{}
    workerCount int32  // 当前活跃Worker数
}

// 全局Worker池
var danmakuPool *DanmakuWorkerPool

// InitDanmakuWorkerPool 初始化Worker池
func InitDanmakuWorkerPool(workerCount int) {
    danmakuPool = &DanmakuWorkerPool{
        jobChan:  make(chan DanmakuJob, 10000), // 带缓冲队列
        quit:     make(chan struct{}),
        workerCount: int32(workerCount),

    for i := 0; i < workerCount; i++ {
        danmakuPool.wg.Add(1)
        go danmakuPool.worker(i)

func (p *DanmakuWorkerPool) worker(id int) {
    defer p.wg.Done()

            p.processJob(job)
        case <-p.quit:
            return
        }
    }

    // WebSocket发送
}

// Submit 提交任务（非阻塞）
func (p *DanmakuWorkerPool) Submit(job DanmakuJob) bool {

        return true
    default:
```

### 2.2 批量处理 + Redis Pipeline

当前Redis操作为逐条写入，高并发时应改为Pipeline批量处理：

```go
// BatchRedisUpdate 批量Redis更新
type BatchRedisUpdate struct {
    pending sync.Map // openId → score增量
    mu      sync.Mutex
    ticker  *time.Ticker
    pipe    *redis.Pipeline
}

// StartBatchUpdater 启动批量更新（每100ms提交一次）
func StartBatchUpdater(client *redis.Client, interval time.Duration) *BatchRedisUpdate {
    b := &BatchRedisUpdate{
        pipe:  client.Pipeline(),
        ticker: time.NewTicker(interval),
    }
    
    go func() {
        for range b.ticker.C {
            b.flush()
        }
    }()
    return b
}

// AddScore 添加分数（不立即写Redis）
func (b *BatchRedisUpdate) AddScore(openId string, score int64) {
    b.mu.Lock()
    defer b.mu.Unlock()
    
    if val, ok := b.pending.Load(openId); ok {
        b.pending.Store(openId, val.(int64)+score)
    } else {
        b.pending.Store(openId, score)
    }
}

// flush 批量提交到Redis
func (b *BatchRedisUpdate) flush() {
    b.mu.Lock()
    defer b.mu.Unlock()
    
    if b.pending == nil || len(b.pending) == 0 {
        return
    }
    
    // 遍历所有pending更新
    var cmds []*redis.StatusCmd
    b.pending.Range(func(key, value interface{}) bool {
        openId := key.(string)
        score := value.(int64)
        
        // 使用ZINCRBY批量更新
        cmds = append(cmds, b.pipe.ZIncrBy(
            fmt.Sprintf("GameRank:Week:%d:%s", platformId, getWeekKey()),
            float64(score),
            openId,
        ))
        b.pending.Delete(key)
        return true
    })
    
    // 执行Pipeline
    if len(cmds) > 0 {
        b.pipe.Exec()
        // 重置Pipeline
        b.pipe = redisClient.Pipeline()
    }
}
```

### 2.3 分片并发Map（针对冷却检查优化）

`sync.Map` 在高频读写场景下性能不如分片Map：

```go
// ShardedCooldown 分片冷却检查
type ShardedCooldown struct {
    shards    []*cooldownShard
    shardMask uint64
    numShards uint64
}

type cooldownShard struct {
    mu    sync.RWMutex
    data  map[string]time.Time // key: openId:action
}

func NewShardedCooldown(numShards uint64) *ShardedCooldown {
    s := &ShardedCooldown{
        numShards:  numShards,
        shardMask:  numShards - 1,
        shards:     make([]*cooldownShard, numShards),
    }
    for i := range s.shards {
        s.shards[i] = &cooldownShard{data: make(map[string]time.Time)}
    }
    return s
}

func (s *ShardedCooldown) getShard(key string) *cooldownShard {
    h := fnvHash(key)
    return s.shards[h&s.shardMask]
}

func (s *ShardedCooldown) Check(key string, cooldownSec int) bool {
    shard := s.getShard(key)
    shard.mu.RLock()
    defer shard.mu.RUnlock()
    
    if lastTime, ok := shard.data[key]; ok {
        if time.Since(lastTime) < time.Duration(cooldownSec)*time.Second {
            return false // 冷却中
        }
    }
    return true
}

func (s *ShardedCooldown) Set(key string) {
    shard := s.getShard(key)
    shard.mu.Lock()
    defer shard.mu.Unlock()
    shard.data[key] = time.Now()
}

// fnvHash FNV哈希（比直接hash更快）
func fnvHash(s string) uint64 {
    h := uint64(14695981039346656037)
    for i := 0; i < len(s); i++ {
        h ^= uint64(s[i])
        h *= 1099511628211
    }
    return h
}
```

|-----|-------|-------|

| ShardedMap | **350万/s** | **200万/s** |

### 3.2 优化方案：Goroutine-Safe Send Channel

```go
type WsClient struct {
    Conn      *websocket.Conn
    sendChan  chan []byte
    sendMutex sync.Mutex
    isClosed  int32       // 原子操作标志
}

// WsSend 异步发送（推荐）
func (c *WsClient) WsSend(cmd int, msg proto.Message) error {
    // 序列化
    data, err := proto.Marshal(msg)
    if err != nil {
        return err
    }
    
    // 封装协议头
    packet := make([]byte, 2+len(data))
    binary.BigEndian.PutUint16(packet[0:2], uint16(cmd))
    copy(packet[2:], data)
    
    // 发送到Channel（不阻塞）
    select {
    case c.sendChan <- packet:
        return nil
    case <-time.After(5 * time.Millisecond):
        // 5ms超时，降级日志
        return fmt.Errorf("send timeout")
    }
}

// sendLoop 专门的发送协程
func (c *WsClient) sendLoop() {
    for packet := range c.sendChan {
        if atomic.LoadInt32(&c.isClosed) == 1 {
            return
        }
        
        c.sendMutex.Lock()
        err := c.Conn.WriteMessage(websocket.BinaryMessage, packet)
        c.sendMutex.Unlock()
        
        if err != nil {
            // 记录错误，触发重连
            c.handleWriteError(err)
            return
        }
    }
}
```

```go
// StartHeartbeat 启动心跳
func (c *WsClient) StartHeartbeat() {
    ticker := time.NewTicker(30 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ticker.C:
            if atomic.LoadInt32(&c.isClosed) == 1 {
                return
            }
            // 发送Ping
            c.sendMutex.Lock()
            err := c.Conn.WriteMessage(websocket.PingMessage, nil)
            c.sendMutex.Unlock()
            
            if err != nil {
                c.handleDisconnect()
                return
            }
        }
    }
}

// handleDisconnect 处理断连
func (c *WsClient) handleDisconnect() {
    atomic.StoreInt32(&c.isClosed, 1)
    close(c.sendChan) // 停止sendLoop
    
    // 重连逻辑
    go c.reconnect()
}

// reconnect 指数退避重连
func (c *WsClient) reconnect() {
    backoff := time.Second
    maxBackoff := 60 * time.Second
    
    for {
        time.Sleep(backoff)
        
        err := c.connect()
        if err == nil {
            return // 重连成功
        }
        
        // 指数退避
        backoff *= 2
        if backoff > maxBackoff {
            backoff = maxBackoff
        }
    }
}
```

```go
// DanmakuLoadTest 弹幕压测客户端
type DanmakuLoadTest struct {

    // 统计

// Run 开始压测
func (t *DanmakuLoadTest) Run() {

    defer ticker.Stop()
    
    for i := 0; i < t.TotalConnections; i++ {
        t.wg.Add(1)
        go t.simulateUser(i, ticker)
    }
    
    t.wg.Wait()
    
    // 输出报告
    t.printReport(time.Since(start))

// simulateUser 模拟单个用户
func (t *DanmakuLoadTest) simulateUser(userId int, ticker *time.Ticker) {
    defer t.wg.Done()
    
    // 建立WebSocket连接
    ws, err := dialWS()
    if err != nil {
        atomic.AddInt64(&t.failCount, 1)
        return
    }
    defer ws.Close()
    
    keywords := []string{"攻击", "防守", "治疗", "加油", "666"}
    
    for {
        select {
        case <-ticker.C:
            msg := fmt.Sprintf(`{"type":"danmu","content":"%s","userId":"user%d"}`,
                keywords[userId%len(keywords)], userId)
            
            sendStart := time.Now()
            err := ws.WriteJSON(msg)
            latency := time.Since(sendStart).Milliseconds()
            
            if err != nil {
                atomic.AddInt64(&t.failCount, 1)
                return

            // 更新最大延迟

                old := atomic.LoadInt64(&t.latencyMax)
                if latency <= old {
                    break
                }
                if atomic.CompareAndSwapInt64(&t.latencyMax, old, latency) {
                    break
                }
            }
        case <-time.After(t.Duration):
            return
        }
    }
}
```
