// 自定义二进制协议
// 协议格式: [type:1byte][length:2bytes][data:nbytes]

export class BinaryProtocol {
    static encode(type: number, data: any): ArrayBuffer {
        const jsonStr = JSON.stringify(data);
        const encoder = new TextEncoder();
        const payload = encoder.encode(jsonStr);
        
        const buffer = new ArrayBuffer(1 + 2 + payload.length);
        const view = new DataView(buffer);
        
        // type
        view.setUint8(0, type);
        // length
        view.setUint16(1, payload.length, false);
        // payload
        for (let i = 0; i < payload.length; i++) {
            view.setUint8(3 + i, payload[i]);
        }
        
        return buffer;
    }
    
    static decode(buffer: ArrayBuffer): {type: number, data: any} {
        const view = new DataView(buffer);
        const type = view.getUint8(0);
        const length = view.getUint16(1, false);
        
        const payload = new Uint8Array(buffer, 3, length);
        const decoder = new TextDecoder();
        const data = JSON.parse(decoder.decode(payload));
        
        return { type, data };
    }
}

    }
    
    // 彩虹色渐变
    setRainbowMode(enabled: boolean) {
        if (enabled) {
            this.schedule(this.updateRainbow, 0.1);
        } else {
            this.unschedule(this.updateRainbow);
        }
    }
    
    private updateRainbow() {
        const hue = (Date.now() / 20) % 360;
        const sprite = this.node.getComponent(Sprite);
        if (sprite) {
            sprite.color = this.hsvToRgb(hue, 1, 1);
        }
    }
    
    private hsvToRgb(h: number, s: number, v: number): Color {
        // HSV转RGB实现
        const f = (n: number, k: number = (n + h / 60) % 6) => 
            v - v * s * Math.max(Math.min(k, 4 - k, 1), 0);
        return new Color(f(5) * 255, f(3) * 255, f(1) * 255, 255);
    }
}

// 对象池优化
export class DanmuPool {
    private pool: Node[] = [];
    private activeDanmus: Set<Node> = new Set();
    
    constructor(private prefab: Prefab, private maxSize: number = 100) {
        // 预创建
        for (let i = 0; i < this.maxSize; i++) {
            const node = instantiate(this.prefab);
            node.active = false;
            this.pool.push(node);
            this.node.addChild(node);
        }
    }
    
    get(): Node | null {
        // 从池中取
        for (const node of this.pool) {
            if (!node.active) {
                node.active = true;
                this.activeDanmus.add(node);
                return node;
            }
        }
        
        // 池已满，扩容
        if (this.pool.length < this.maxSize * 2) {
            const node = instantiate(this.prefab);
            this.pool.push(node);
            this.node.addChild(node);
            node.active = true;
            this.activeDanmus.add(node);
            return node;
        }
        
        return null;
    }
    
    recycle(node: Node) {
        node.active = false;
        this.activeDanmus.delete(node);
    }
    
    // 批量回收
    recycleAll() {
        this.activeDanmus.forEach(node => {
            node.active = false;
        });
        this.activeDanmus.clear();
    }
}

        if (window.performance && (performance as any).memory) {
            const memory = (performance as any).memory;
            return {
                js: memory.usedJSHeapSize,
                native: memory.totalJSHeapSize - memory.usedJSHeapSize,
                total: memory.totalJSHeapSize
            };
        }
        return { js: 0, native: 0, total: 0 };
    }

    }

    forceGC() {
        if (window.gc) {
            window.gc();

        }

// 游戏状态同步
export class GameSync {
    private serverState: GameState;
    private clientState: GameState;
    private pendingInputs: Input[] = [];
    
    // 本地预测
    predict(input: Input) {
        this.pendingInputs.push(input);
        this.clientState = this.applyInput(this.clientState, input);
        this.render(this.clientState);
    }
    
    // 服务器校正
    reconcile(serverState: GameState, inputs: Input[]) {
        // 移除已确认的输入
        this.pendingInputs = this.pendingInputs.filter(
            input => !inputs.includes(input)
        );
        
        // 重放剩余输入
        let predictedState = serverState;
        for (const input of this.pendingInputs) {
            predictedState = this.applyInput(predictedState, input);
        }
        
        // 误差检测
        if (this.getDistance(predictedState, this.clientState) > 0.1) {
            // 客户端与服务器差异过大，进行平滑校正
            this.smoothCorrect(predictedState);
        }
    }
    
    private applyInput(state: GameState, input: Input): GameState {
        // 根据输入更新状态
        return state;
    }
    
    private getDistance(a: GameState, b: GameState): number {
        return Math.abs(a.playerX - b.playerX);
    }
    
    private smoothCorrect(targetState: GameState) {
        // 平滑过渡到正确状态
        tween(this.clientState)
            .to(0.1, targetState)
            .start();
    }
}

// 指令解析器
export class CommandParser {
    private commands: Map<string, CommandHandler> = new Map();
    
    constructor() {
        // 注册指令
        this.register('修炼', this.handleCultivate);
        this.register('境界', this.handleRealm);
        this.register('灵石', this.handleSpiritStones);
        this.register('法宝', this.handleArtifact);
        this.register('渡劫', this.handleTribulation);
    }
    
    parse(text: string): CommandResult {
        for (const [cmd, handler] of this.commands) {
            if (text.startsWith(cmd)) {
                const args = text.slice(cmd.length).trim();
                return handler(args);
            }
        }
        
        // 默认当作普通弹幕（增加修炼值）
