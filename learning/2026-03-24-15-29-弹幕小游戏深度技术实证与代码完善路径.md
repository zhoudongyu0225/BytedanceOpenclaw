```go
type WebsocketAnchorClient struct {
    Socket             *websocket.Conn
    Send               chan []byte        // 写通道(带缓冲10000)
    BarrageMessageChan chan *pb.MessageList // 弹幕接收通道
```

```go
func GiftMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    for _, msg := range messageList.MsgList {
        if msg.Uid == "" {
            continue
        }

        pdb, _ := GetOpenVo(msg.Uid)
        if pdb == nil {
            continue
        }

        // 1. 获取礼物配置
        giftConfig := GetGiftConfig(msg.GiftId) // 从Buff.json/Gift.json加载
        if giftConfig == nil {
            zlog.Logger.Error().Msgf("礼物配置不存在:%v", msg.GiftId)
            continue
        }

        // 2. 计算礼物价值
        totalValue := computeGiftValue(msg.GiftId, msg.GiftCount)

        // 3. 计算游戏内效果
        // 士兵数 = 礼物价值 / 配置系数
        soldierNum := computeSoldierNum(totalValue, giftConfig)
        // 治疗值 = 礼物价值 * 治疗系数
        rescueNum := computeRescueNum(totalValue, giftConfig)

        // 4. 更新积分(累加到玩家Score)
        newScore := pdb.Score + int32(totalValue)
        pdb.Score = newScore

        // 5. 更新周榜/月榜
        model.UpdateGameRank(pdb.OpenId, ws.AnchorV.PlatformId, float64(totalValue))

        // 6. 记录本次游戏礼物价值
        ws.giftV += totalValue

        // 7. 广播给客户端(GAME_GIFT前缀)
        ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
            Type:    5, // GameGift
            Content: buildGameGiftPayload(msg, soldierNum, rescueNum),
        })
    }
}
```

### 3.3 LikeMessage点赞逻辑(完整代码框架)

```go
// LikeMessage - 点赞消息处理(1秒Tick触发)
// TODO位置: control_barrage.go ~116行
func LikeMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    if messageList == nil || len(messageList.MsgList) == 0 {
        return
    }

    for _, msg := range messageList.MsgList {
        if msg.Uid == "" {
            continue
        }

        pdb, _ := GetOpenVo(msg.Uid)
        if pdb == nil {
            continue
        }

        // 1. 获取点赞数
        likeCount := untils.StringToInt32(msg.Count)
        if likeCount <= 0 {
            continue
        }

        // 2. 点赞冷却检查
        coolKey := fmt.Sprintf("LikeCool:%s:%s", ws.AccountId, msg.Uid)
        redisCli := model.GetRedisClient()
        exists, _ := redisCli.Exists(ctx, coolKey).Result()
        if exists > 0 {
            continue
        }

        // 3. 累计点赞(Redis原子累加)
        accumKey := fmt.Sprintf("LikeAccum:%s:%s", ws.AccountId, msg.Uid)
        newAccum, _ := redisCli.IncrBy(ctx, accumKey, int64(likeCount)).Result()
        redisCli.Expire(ctx, accumKey, 60*time.Second) // 60秒过期

        // 4. 设置冷却(5秒)
        redisCli.SetNX(ctx, coolKey, "1", 5*time.Second)

        // 5. Buff阈值检查(10/20/50/100)
        if shouldTriggerBuff(newAccum) {
            buffLevel := getBuffLevel(newAccum)
            // 广播Buff触发(GAME_BUFF前缀)
            ws.WsSend(COMMAND_TIPS_S2C, &pb.TipsNotify{
                Type:    6, // GameBuff
                Content: buildGameBuffPayload(msg, buffLevel),
            })
        }
    }
}
```
