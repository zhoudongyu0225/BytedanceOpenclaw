### 7.1 VeADK-Go 弹幕审核Agent 最简代码

```go
package main

import (
	"context"
	"fmt"
	"os"

	veagent "github.com/volcengine/veadk-go/agent/llmagent"
	"github.com/volcengine/veadk-go/log"
	"google.golang.org/adk/agent"
	"google.golang.org/adk/cmd/launcher"
	"google.golang.org/adk/cmd/launcher/full"
	"google.golang.org/adk/session"
)

func main() {
	ctx := context.Background()

	// 弹幕审核子Agent
	auditAgent, err := veagent.New(&veagent.Config{
		Config: llmagent.Config{
			Name:        "danmaku_audit",
			Description: "弹幕内容安全审核",
			Instruction: `判断弹幕内容是否违规。
返回格式：PASS 表示通过，BLOCK 表示拦截。
拦截原因：敏感词、人身攻击、广告、低俗等。`,
		},
		ModelName: "doubao-seed-1-6-250615",
	})
	if err != nil {
		fmt.Printf("NewLLMAgent auditAgent failed: %v", err)
		return
	}

	// 根Agent
	rootAgent, err := veagent.New(&veagent.Config{
		Config: llmagent.Config{
			Name:        "root",
			Description: "恐龙塔防弹幕游戏AI中枢",
			SubAgents:   []agent.Agent{auditAgent},
		},
		ModelName: "doubao-seed-1-6-250615",
	})
	if err != nil {
		fmt.Printf("NewLLMAgent rootAgent failed: %v", err)
		return
	}

	config := &launcher.Config{
		AgentLoader:    agent.NewSingleLoader(rootAgent),
		SessionService: session.InMemoryService(),
	}

	l := full.NewLauncher()
	if err = l.Execute(ctx, config, os.Args[1:]); err != nil {
		log.Fatalf("Run failed: %v\n\n%s", err, l.CommandLineSyntax())
	}
}
```

### 7.2 弹幕→游戏指令 规则引擎设计（伪代码）

```go
// 规则定义
type Rule struct {
    Keywords []string
    Action   GameAction
    Cooldown int // 冷却时间(秒)
}

// 游戏动作
type GameAction struct {
    Type    string // "spawn_dinosaur", "upgrade_tower", etc.
    Params  map[string]interface{}
}

// 规则引擎
type RuleEngine struct {
    rules    []Rule
    cooldowns map[string]int64 // playerID_action -> last_trigger_time
}

func (e *RuleEngine) Process(text string, playerID string) *GameAction {
    text = strings.ToLower(text)
    for _, rule := range e.rules {
        for _, kw := range rule.Keywords {
            if strings.Contains(text, kw) {
                // 检查冷却
                key := playerID + "_" + rule.Action.Type
                if last, ok := e.cooldowns[key]; ok {
                    if time.Now().Unix()-last < int64(rule.Cooldown) {
                        return nil // 冷却中
                    }
                }
                e.cooldowns[key] = time.Now().Unix()
                return &rule.Action
            }
        }
    }
    return nil
}

// 预设规则
var defaultRules = []Rule{
    {
        Keywords: []string{"出龙", "龙", "dino", "恐龙"},
        Action:   GameAction{Type: "spawn_dinosaur", Params: map[string]interface{}{"tier": 1}},
        Cooldown: 5,
    },
    {
        Keywords: []string{"升级", "升", "up", "强"},
        Action:   GameAction{Type: "upgrade_tower", Params: map[string]interface{}{"amount": 1}},
        Cooldown: 10,
    },
    // ... 更多规则
}
```
