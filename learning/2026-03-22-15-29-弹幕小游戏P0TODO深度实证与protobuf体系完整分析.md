### 2.3 TipMsg 前缀协议设计 (经学习验证)
```
TipMsg 前缀格式: "{TYPE}|{数据}"

GAME_ACTION|{openId}|{nickName}|{弹幕内容}     # 弹幕消息
GAME_GIFT|{openId}|{nickName}|{礼物Id}|{礼物名}  # 礼物消息
GAME_BUFF|{openId}|{nickName}|{buffId}|{描述}   # Buff触发消息
```

UE4客户端按 `|` 分隔解析，第一个字段判断消息类型。

### 2.4 TipType 扩展方案
**方案A (推荐)**: 扩展 protobuf
```protobuf
// Z4_TipsNotify.proto
enum TipType {

```


```go
func ChatMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    for _, msg := range messageList.MsgList {
        // ... 现有玩家信息和排行获取代码 ...

        // ========== TODO:弹幕逻辑 ==========
        
        // 第1步: 敏感词过滤
        sf := untils.GetSensitiveFilter()
        if sf.ContainsSensitiveWord(msg.Content) {
            zlog.Logger.Info().Msgf("弹幕含敏感词过滤: %s", msg.Content)
            cleanContent := sf.ReplaceSensitiveWords(msg.Content, '*')
            if cleanContent == "" {
                continue // 完全敏感,丢弃
            }
            msg.Content = cleanContent
        }
        
        // 第2步: 冷却检查 (同用户弹幕冷却 5秒)
        cooldownKey := fmt.Sprintf("Cooldown.Chat.%s.%s", msg.Uid, ws.AccountId)
        exists, _ := model.GetRedisClient().Exists(context.Background(), cooldownKey).Result()
        if exists > 0 {
            continue // 冷却中,丢弃
        }
        model.GetRedisClient().Set(context.Background(), cooldownKey, "1", 5*time.Second)
        
        // 第3步: 积分更新 (每条弹幕 +1分)
        scoreDelta := int64(1)
        model.UpdateGameWeekRank(msg.Uid, scoreDelta, ws.AnchorV.PlatformId)
        model.UpdateGameMonthRank(msg.Uid, scoreDelta, ws.AnchorV.PlatformId)
        
        // 第4步: 广播到UE4客户端 (TipsNotify)
        tipMsg := fmt.Sprintf("GAME_ACTION|%s|%s|%s",
            msg.Uid, msg.Name, msg.Content)
        ws.WsSend(COMMAND_TIPS_NOTIFY_S, &pb.TipsNotify{
            TipMsg: tipMsg,
            Type:   pb.TipType_RoomRoll, // 或扩展后的GameAction
        })
    }
}
```

```go
func LikeMessage(messageList *pb.MessageList, ws *WebsocketAnchorClient) {
    now := time.Now().Unix()
    msgJoin(messageList, ws)
    for _, msg := range messageList.MsgList {
        // ... 现有代码 ...
        
        // ========== TODO:点赞逻辑 ==========
        
        // 第1步: 解析点赞数 (使用 msg.GetCount() 字段)
        // ⚠️ 重要发现: Message protobuf 有专门的 Count 字段 (field #6) 用于点赞数量
        //    而不是解析 Content 字段
        likeCount, _ := strconv.ParseInt(msg.GetCount(), 10, 64)
        if likeCount <= 0 {
            likeCount = 1
        }
        
        // 第2步: 更新玩家累计点赞
        likeTotalKey := fmt.Sprintf("Player.LikeTotal.%s.%s", msg.Uid, ws.AccountId)
        newTotal, _ := model.GetRedisClient().IncrBy(context.Background(),
            likeTotalKey, int64(likeCount)).Result()
        
        // 第3步: 检查Buff触发 (点赞阈值Buff: BuffType=1)
        // BuffId=1: 阈值10, BuffId=2: 阈值5, BuffId=3: 阈值3
        triggeredBuffs := checkLikeBuffTrigger(newTotal)
        for _, buffId := range triggeredBuffs {
            buffCfg := config.GetBuffConfig(float64(buffId))
            if buffCfg == nil {
                continue
            }
            
            // 获取Buff奖励士兵数
            rewardSoldiers := int64(0)
            if len(buffCfg.IntArray_Param) > 0 {
                rewardSoldiers = int64(buffCfg.IntArray_Param[0])
            }
            
            // 更新积分
            model.UpdateGameWeekRank(msg.Uid, rewardSoldiers, ws.AnchorV.PlatformId)
            model.UpdateGameMonthRank(msg.Uid, rewardSoldiers, ws.AnchorV.PlatformId)
            
            // 广播Buff触发
            tipMsg := fmt.Sprintf("GAME_BUFF|%s|%s|%d|%s",
                msg.Uid, msg.Name, buffId, fmt.Sprintf("点赞达标奖励%d士兵", rewardSoldiers))
            ws.WsSend(COMMAND_TIPS_NOTIFY_S, &pb.TipsNotify{
                TipMsg: tipMsg,
                Type:   pb.TipType_RoomRoll,
            })
        }
        
        // 第4步: 点赞积分 (点赞本身也加分)
        model.UpdateGameWeekRank(msg.Uid, int64(likeCount), ws.AnchorV.PlatformId)
        model.UpdateGameMonthRank(msg.Uid, int64(likeCount), ws.AnchorV.PlatformId)
    }
}

// 解析点赞数 (需要根据实际平台协议调整)
func parseLikeCount(content string) int64 {
    if content == "" {
        return 1
    }
    // 格式1: "like:100" → 提取100
    if strings.HasPrefix(content, "like:") {
        n, _ := strconv.ParseInt(strings.TrimPrefix(content, "like:"), 10, 64)
        return n
    }
    // 格式2: 纯数字
    n, err := strconv.ParseInt(content, 10, 64)
    if err == nil {
        return n
    }
    return 1
}

// 检查点赞Buff触发
func checkLikeBuffTrigger(totalLikes int64) []int32 {
    var triggered []int32
    // BuffId=1: 阈值10, BuffId=2: 阈值5, BuffId=3: 阈值3
    thresholds := []struct {
        buffId   int32
        threshold int64
    }{
        {1, 10},
        {2, 5},
        {3, 3},
    }
    for _, t := range thresholds {
        if totalLikes >= t.threshold {
            triggered = append(triggered, t.buffId)
        }
    }
    return triggered
}
```
